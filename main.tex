% Slide 5: Symmetry of QCD (QCD的对称性)
% ==============================================================================
\begin{frame}
\myheading{Symmetry of QCD}
\vfill
\[
\mathcal{Z} = \int \mathcal{D} A\, \mathcal{D}\bar{q}\, \mathcal{D}q\; e^{-S}, \qquad
S = \int d^4x \left[ - \frac{1}{2g^2} \mathrm{Tr} F^{\mu\nu} F_{\mu\nu}
+ \bar q_a (i \slashed{D} - m_a) q_a \right]
\]
\vspace{0.8em}

For the limit of $m_{u,d} \to 0$:

\vspace{0.4em}
$SU(2)_L \times SU(2)_R$ chiral symmetry:
\[
    q \to \exp(i\alpha_a T^a + i\beta_a T^a \gamma_5)q, \quad \bar{q} \to \bar{q} \exp(-i\alpha_a T^a + i\beta_a T^a \gamma_5)
\]

$U(1)_A$ symmetry (anomalous):
\[
    q \to \exp(i\epsilon\gamma_5)q, \quad \bar{q} \to \bar{q} \exp(i\epsilon\gamma_5)
\]

\vfill
\end{frame}

% ==============================================================================
\begin{frame}
\myheading{Meson Correlation Functions}
\vfill
Flavor non-singlet meson operators:
\begin{gather*}
    O_\Gamma(x) = \bar{q}(x)\Gamma F^aq(x), \\[2ex]
    \Gamma \in \left\{ \mathbf{1}(S),\; \gamma_5(PS),\; \gamma_\mu(V),\; \gamma_\mu\gamma_5(A),\; \sigma_{\mu\nu}(T),\; \sigma_{\mu\nu}\gamma_5(X) \right\}
\end{gather*}

\vspace{0.5em}
Spatial correlators define the screening mass $m_\Gamma$:
\[
    C_\Gamma(x) \simeq A \cosh\!\left(m_\Gamma\left(x - \frac{N_s}{2}\right)\right)
\]
\vfill
\end{frame}

\begin{itemize}

    \item Average over 16 sources and spatial directions $(x, y, z)$
    \item Effective mass to choose the plateau region:
    \[
        \frac{C(x)}{C(x + 1)} = \frac{\cosh(m_{\text{eff}}(x - N_s/2))}{\cosh(m_{\text{eff}}(x + 1 - N_s/2))}
    \]
    \item $\cosh$ fit in the plateau, errors from jackknife:
    \[
        C(x) = A \cosh\left(m_{\text{scr}}\left(x - \frac{N_s}{2}\right)\right)
    \]
\end{itemize}

\begin{frame}[noframenumbering]
\myheading{Backup: Overlap Dirac Operator}
\vfill
The massless operator $D_{\mathrm{ov}}(0) = \frac{1}{2} (1 + \gamma_5 \mathrm{sgn}(H_W))$ with Hermitian $H_W$ satisfies the Ginsparg-Wilson relation:
\[ \gamma_5 D_{\mathrm{ov}} + D_{\mathrm{ov}} \gamma_5 = 2 D_{\mathrm{ov}} \gamma_5 D_{\mathrm{ov}} \]
The massive overlap operator is defined as:
\[ D_{\mathrm{ov}}(m) = D_{\mathrm{ov}}(0) + m (1 - D_{\mathrm{ov}}(0)) = \frac{1+m}{2} + \frac{1-m}{2} \gamma_5 \mathrm{sgn}(H_W) \]
\begin{equation}
\widehat{\gamma}_5 = \gamma_5 \left( 1 - 2 \slashed{D}_{\mathrm{ov}}(0) \right) = -\gamma_5 \mathrm{sgn}(H_W)
\label{eq:hat_gamma_5}
\end{equation}
which is hermitian and satisfies $\widehat{\gamma}_5^2 = 1$ by virtue of the Ginsparg-Wilson relation. The lattice chiral transformation is defined as:
\begin{equation}
\delta \psi = i \alpha \widehat{\gamma}_5 \psi, \quad \delta \overline{\psi} = i \alpha \overline{\psi} \gamma_5
\label{eq:overlap_chiral_trans}
\end{equation}
\vfill
\end{frame}

\vfill
\setbeamerfont{itemize/enumerate body}{size=\fontsize{9pt}{11.5pt}\selectfont}
\fontsize{9pt}{11.5pt}\selectfont
\begin{itemize}
    \item Perturbative $4-\epsilon$ expansion RG analysis of effective $\Phi^4$ linear sigma model shows no stable IR fixed point for $N_f > \sqrt{3}$ flavor symmetry {\scriptsize [Pisarski-Wilczek (1984)]}.
    \item For $N_f \geq 3$, the anomaly determinant term doesn't change the order of the transition; for $N_f=2$, it becomes a quadratic mass-like term which can smooth out the singularity and make the transition 2nd-order.
    \item \textbf{Previous Scenario}: 1st-order if $U(1)_A$ is restored at $T_c$; 2nd-order $\mathrm{O}(4)$ if $U(1)_A$ remains broken.

\end{itemize}

\begin{frame}[noframenumbering]
\myheading{Backup: Phenomenology of Symmetry Restoration}
\vfill
\setbeamerfont{itemize/enumerate body}{size=\fontsize{8.5pt}{11pt}\selectfont}
\fontsize{8.5pt}{11pt}\selectfont
\begin{itemize}
    \setlength{\itemsep}{0.6em}
    \item \textbf{Chiral Symmetry ($SU(2)_L \times SU(2)_R$) \& Condensate}:
    \begin{itemize}
        \item GMOR relation connects pseudo-Goldstone pion mass and chiral condensate:
        \[ m_\pi^2 f_\pi^2 = -m_q \langle 0 | \bar{q}q | 0 \rangle \]
        \item Chiral susceptibility $\chi_{\mathrm{disc}} = \frac{\partial \langle \bar{q}q \rangle}{\partial m_l} = \frac{1}{4}(\chi_\pi - \chi_\delta)$ peaks at $T_{pc}$, where $\langle \bar{q}q \rangle \to 0$.
        \item Condensate melting removes spontaneous chiral breaking $\Rightarrow$ parity pairs degenerate: $V \leftrightarrow A$, $S \leftrightarrow PS$.
    \end{itemize}
    \item \textbf{Axial $U(1)_A$ Anomaly \& Topological Susceptibility}:
    \begin{itemize}
        \item Witten-Veneziano relation links $\eta'$ mass to topological susceptibility $\chi_t$:
        \[ F_\pi^2 m_{\eta'}^2 = 2 N_f \chi_t, \qquad \chi_t = \int d^4x\, \langle q(x) q(0) \rangle \propto n_I(T) \]
        \item Debye screening suppresses instantons at high $T$:
        $n(\rho, T) \propto \exp\left( -\frac{1}{3}(2N_c + N_f)(\pi \rho T)^2 \right)$.
        \item As $T > T_c$, $n_I(T) \to 0 \implies \chi_t \to 0$, restoring $U(1)_A$ symmetry ($S \leftrightarrow PS$, $T_t \leftrightarrow X_t$).
    \end{itemize}
    \item \textbf{High-$T$ Spatial Screening Mass (3D-NRQCD)}:
    \begin{itemize}
        \item Lowest Matsubara mode $p_0 = \pi T$ gives spin/parity-independent screening mass:
        $m_{\mathrm{scr}} \approx 2\pi T + 0.1872\, g^2 T$.
    \end{itemize}
\end{itemize}
\vfill
\end{frame}
