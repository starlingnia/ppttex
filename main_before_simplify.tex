\myheading{Research Background}
\setbeamerfont{itemize/enumerate body}{size=\fontsize{10.5pt}{15pt}\selectfont}
\fontsize{10.5pt}{15pt}\selectfont
\vfill
\begin{itemize}
\setlength{\itemsep}{0.7em}
\item According to \textbf{Pisarski \& Wilczek (1984)}, the restoration of $U(1)_A$ axial symmetry around the chiral phase transition is crucial for determining the order and universality class of the transition.
\item Previous lattice studies often relied on staggered or Wilson-type fermions that explicitly break chiral or taste symmetries at finite lattice spacing $a$.
\item Our recent $N_f = 2$ study by the \textbf{JLQCD Collaboration} {\scriptsize [Phys. Rev. D \textbf{111}, 114506 (2025)]} using Möbius Domain-Wall Fermions (MDWF) showed that both $SU(2)_L \times SU(2)_R$ chiral symmetry and $U(1)_A$ symmetry are effectively restored near $T_c \approx 165(3)$ MeV.
\end{itemize}
\vfill
\end{frame}

% Slide 5: Symmetry of QCD (QCD的对称性)
% ==============================================================================
\begin{frame}
\myheading{Symmetry of QCD}
\vfill
QCD Partition Function: \hfill
 $\mathcal{Z} =
\int \mathcal{D} A \mathcal{D}\bar{q} \mathcal{D}qe^{-S}$, \quad Gauge Boson + Fermion\hfill\null

\vspace{0.8em}{
\centering
$S =
\int d^4x \left[ - \frac{1}{2g^2} \mathrm{Tr} F^{\mu\nu} F_{\mu\nu}
+ \bar q_a (i \slashed{D} - m_a) q_a \right]$, \quad $a \in [1, N_f]$
}
\vspace{0.8em}

In the massless up-down quark limit ($m_{u,d} \to 0$):

Flavored chiral symmetry $SU(2)_L \times SU(2)_R$ :
\[
    q \to \exp(i\alpha_a T^a + i\beta_a T^a \gamma_5)q, \quad \bar{q} \to \bar{q} \exp(-i\alpha_a T^a + i\beta_a T^a \gamma_5),
\]

$U(1)_A$ Axial Symmetry (broken at the quantum level by the ABJ anomaly):
\[
    q \to \exp(i\epsilon\gamma_5)q, \quad \bar{q} \to \bar{q} \exp(i\epsilon\gamma_5).
\]

\vfill
\end{frame}

% ==============================================================================
% Slide 6: Possible Emergent Larger Symmetry (可能涌现的更大对称性 - 手征自旋对称性)
% ==============================================================================
\begin{frame}
\myheading{Possible Emergent Larger Symmetry}
\setbeamerfont{itemize/enumerate body}{size=\fontsize{10pt}{13pt}\selectfont}
\fontsize{10pt}{13pt}\selectfont
\vfill

\begin{itemize}
\setlength{\itemsep}{0.5em}
\item Dimensional reduction / $\mathrm{NRQCD}_3$ perturbative analysis {\scriptsize{[Laine \& Vepsäläinen (2004)]}} predicts a \textbf{spin-independent} asymptotic screening mass at high $T$ due to the large Matsubara mass $M \approx \pi T$:
\[ m_{\mathrm{scr}} = 2\pi T + g_E^2 \frac{C_F}{2\pi}\left(\frac{1}{2} + \hat{E}_0\right) \approx 2\pi T + 0.1872 g^2 T \]
\item $U(1)_A$ is expected to be restored at high temperature as instanton/topological fluctuations are suppressed.
\item Furthermore, an emergent larger symmetry, \textbf{Chiral-Spin Symmetry $SU(2)_{CS}$}, has been proposed in the regime where chromoelectric interactions dominate over chromomagnetic ones:
\begin{align*}
    q &\to \exp(i\Sigma^i\epsilon_i)q, \\
    \bar{q} &\to \bar{q}\gamma_0 \exp(-i\Sigma^i\epsilon_i)\gamma_0,& \text{where } \Sigma^i = (\gamma_k, -i\gamma_5\gamma_k, \gamma_5), \quad k = 1, 2.
\end{align*}
\end{itemize}
\vfill
\end{frame}

% Slide 7: Meson Correlation Functions (介子关联函数)
% ==============================================================================
\begin{frame}
\myheading{Meson Correlation Functions}
\vfill
We consider the following set of flavor non-singlet meson operators:
\begin{gather*}
    O_\Gamma(x) = \bar{q}(x)\Gamma F^aq(x), \\[2ex]
    \Gamma \in \left\{ \mathbf{1}(S),\; \gamma_5(PS),\; \gamma_\mu(V),\; \gamma_\mu\gamma_5(A),\; \sigma_{\mu\nu} = \frac{i}{2}[\gamma_\mu, \gamma_\nu](T),\; \sigma_{\mu\nu}\gamma_5(X) \right\}
\end{gather*}

We measure the spatial two-point correlation functions along spatial directions ($x, y, z$).
By periodicity on a lattice of spatial extent $L = N_s a$, the asymptotic behavior at large spatial separation is governed by the screening mass $m_\Gamma$:
\[
    C_\Gamma(x) \simeq A \cosh\!\left(m_\Gamma\left(x - \frac{N_s}{2}\right)\right)
\]
\vfill
\end{frame}

\setbeamerfont{itemize/enumerate body}{size=\fontsize{8.5pt}{11pt}\selectfont}
\fontsize{8.5pt}{11pt}\selectfont
\begin{itemize}
    \item To eliminate additive UV divergences $\propto m_q / a^2$ and residual chiral breaking $m_{\mathrm{res}}$, we construct the renormalized subtracted condensate evaluated with $Z_2$ stochastic volume sources:
    \[
        \Delta_{\ell, s}^R(T) = \frac{1}{Z_m(\beta)} \left[ \langle\bar{\psi}\psi\rangle_\ell - \frac{m_\ell + m_{\mathrm{res}}}{m_s + m_{\mathrm{res}}} \langle\bar{\psi}\psi\rangle_s \right], \qquad \langle\bar{\psi}\psi\rangle_{\mathrm{phys}} = (N_t T)^3 \Delta_{\ell, s}^R(T)
    \]
\end{itemize}

% ==============================================================================
% Slide 14: Extracting Screening Masses: Methodology (抽取筛选质量：方法学)
% ==============================================================================
\begin{frame}
\myheading{Extracting Screening Masses: Methodology}
\vfill
\setbeamerfont{itemize/enumerate body}{size=\fontsize{8.5pt}{11.5pt}\selectfont}
\begin{itemize}
    \item \textbf{Multi-Source \& Directional Averaging}: Correlators are averaged over 16 spatial sources and all equivalent polarization/spatial directions ($x,y,z$), preserving signed correlators prior to folding around $N_s/2$.
    \item \textbf{Local Effective Mass}: Solved numerically via vectorized Newton-Raphson at each separation $x$ to identify a flat ground-state plateau window $[x_{\text{min}}, x_{\text{max}}]$:
    \[
        \frac{C(x)}{C(x + 1)} = \frac{\cosh(m_{\text{eff}}(x)(x - N_s/2))}{\cosh(m_{\text{eff}}(x)(x + 1 - N_s/2))}
    \]
    \item \textbf{Screening Mass Fit \& Jackknife}: $m_{\text{scr}}$ is extracted by a least-squares $\chi^2$ fit on each binned Jackknife sample to eliminate HMC autocorrelation bias:
    \[
        C(x) = A \cosh\left(m_{\text{scr}}\left(x - \frac{N_s}{2}\right)\right)
    \]
\end{itemize}
\vfill
\end{frame}

\begin{frame}[noframenumbering]
\myheading{Backup: Overlap Dirac Operator}
\vfill
The massless operator $D_{\mathrm{ov}}(0) = \frac{1}{2} (1 + \gamma_5 \mathrm{sgn}(H_W))$ with Hermitian $H_W$ satisfies the Ginsparg-Wilson relation:
\[ \gamma_5 D_{\mathrm{ov}} + D_{\mathrm{ov}} \gamma_5 = 2 D_{\mathrm{ov}} \gamma_5 D_{\mathrm{ov}} \]
The massive overlap operator is defined as:
\[ D_{\mathrm{ov}}(m) = D_{\mathrm{ov}}(0) + m (1 - D_{\mathrm{ov}}(0)) = \frac{1+m}{2} + \frac{1-m}{2} \gamma_5 \mathrm{sgn}(H_W) \]
\begin{equation}
\widehat{\gamma}_5 = \gamma_5 \left( 1 - 2 \slashed{D}_{\mathrm{ov}}(0) \right) = -\gamma_5 \mathrm{sgn}(H_W)
\label{eq:hat_gamma_5}
\end{equation}
which is hermitian and satisfies $\widehat{\gamma}_5^2 = 1$ by virtue of the Ginsparg-Wilson relation. The lattice chiral transformation is defined as:
\begin{equation}
\delta \psi = i \alpha \widehat{\gamma}_5 \psi, \quad \delta \overline{\psi} = i \alpha \overline{\psi} \gamma_5
\label{eq:overlap_chiral_trans}
\end{equation}
\vfill
\end{frame}
